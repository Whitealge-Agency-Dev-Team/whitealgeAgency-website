 Middlewares

 AuthToken Middleware
// Verificación JWT
const authToken = (req, res, next) => {
  const authHeader = req.headers["authorization"];
  const token = authHeader && authHeader.split(" ")[1];
  
  if (!token) return res.status(401).json({ message: "Credenciales no válidas" });
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded;
    next();
  } catch (err) {
    return res.status(403).json({ message: "Token inválido" });
  }
};

 AuthRole Middleware
// Verificación de permisos por rol
const authRole = (model, action) => {
  return (req, res, next) => {
    if (!req.user || !req.user.role) {
      return res.status(401).json({ message: "Acceso denegado" });
    }
    
    const userRole = req.user.role;
    const rolePermissions = CRM_PERMISSIONS[userRole];
    const hasPermission = rolePermissions[model]?.includes(action);
    
    if (hasPermission) next();
    else return res.status(403).json({ 
      message: `Rol ${userRole} no tiene permisos para ${action} en ${model}` 
    });
  };
};

Flujos de Trabajo

 1. Flujo de Captación de Clientes
1. POST /clients/public-form (Público)
2. Organizador revisa la problemática
3. PUT /clients/:id/status (Aceptar/Rechazar)
4. Si es aceptado: POST /auth/register-client
5. Cliente recibe credenciales por email
6. Cliente accede con GET /auth/login

 2. Flujo de Gestión de Proyectos
1. POST /projects (Crear proyecto)
2. POST /projects/:id/team (Asignar equipo)
3. POST /projects/:id/objectives (Definir objetivos)
4. POST /projects/:id/key-dates (Establecer hitos)
5. Cliente monitorea con GET /clients/:id/projects

 3. Flujo de Gestión de Personal
1. POST /auth/register-worker (Nuevo empleado)
2. POST /users/:id/salaries (Registrar salario)
3. PUT /users/:id/projects (Asignar a proyectos)
4. POST /users/:id/user-dimiss (Baja si aplica)
